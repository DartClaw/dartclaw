Here's the new JWT middleware:

import jwt from 'jsonwebtoken';

export function verifyToken(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  if (!token) return res.status(401).json({ error: 'No token' });

  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch {
    res.status(403).json({ error: 'Invalid token' });
  }
}

Key changes:

  • Replaced session-based auth with stateless JWT tokens
  • Added verifyToken middleware for protected routes
  • Token is extracted from the Authorization: Bearer header
  • Errors return proper 401/403 status codes
