# Indirect consumers whose accessor is intentionally confined to the config declaration layer.
alerts.routes.<name>  # AlertsConfig.routes selects targets in AlertRouter._resolveTargets.
channels.debounce_window_ms  # ChannelConfig converts milliseconds to debounceWindow, consumed by MessageQueue.
channels.google_chat.quote_reply  # GoogleChatConfig maps this to quoteReplyMode, consumed by GoogleChatChannel and ChannelWiring.
channels.retry_policy.base_delay_ms  # RetryPolicy maps milliseconds to defaultRetryPolicy.baseDelay, consumed by MessageQueue.
channels.<name>  # ChannelConfig.channelConfigs supplies raw named definitions to ChannelConfigResolver.
governance.admin_senders  # GovernanceConfig.isAdmin reads this for the /stop reserved-command admin gate.
guards.command.extra_blocked_patterns  # CommandGuardConfig.fromYaml compiles these into CommandGuard destructivePatterns.
guards.command.extra_blocked_pipe_targets  # CommandGuardConfig.fromYaml merges these into CommandGuard blockedPipeTargets.
guards.network.extra_allowed_domains  # NetworkGuardConfig.fromYaml merges these into NetworkGuard allowedDomains.
guards.network.extra_exfil_patterns  # NetworkGuardConfig.fromYaml compiles these into NetworkGuard exfilPatterns.
mcp_servers.<name>.rate_limit.window_seconds  # MCP parsing maps this to McpServerRateLimit.window, consumed by OutboundMcpPool.
mcp_servers.<name>.token_budget.window_seconds  # MCP parsing maps this to McpServerTokenBudget.window, consumed by OutboundMcpPool.
